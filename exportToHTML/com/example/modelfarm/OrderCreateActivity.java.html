1    package com.example.modelfarm;
2    
3    import android.os.Bundle;
4    import android.view.View;
5    import android.widget.ArrayAdapter;
6    import android.widget.AutoCompleteTextView;
7    import android.widget.Toast;
8    
9    import androidx.activity.EdgeToEdge;
10   import androidx.appcompat.app.AppCompatActivity;
11   
12   import com.example.modelfarm.network.RetrofitClient;
13   import com.example.modelfarm.network.models.ApiResponse;
14   import com.example.modelfarm.network.models.CreateOrderRequest;
15   import com.example.modelfarm.network.models.OrderStatus;
16   import com.example.modelfarm.network.services.OrderApiService;
17   import com.google.android.material.appbar.MaterialToolbar;
18   import com.google.android.material.button.MaterialButton;
19   import com.google.android.material.textfield.TextInputEditText;
20   
21   import retrofit2.Call;
22   import retrofit2.Callback;
23   import retrofit2.Response;
24   
25   public class  OrderCreateActivity extends AppCompatActivity {
26   
27       private MaterialToolbar toolbar;
28       private TextInputEditText etTitle;
29       private TextInputEditText etDescription;
30       private AutoCompleteTextView spStatus;
31       private MaterialButton btnSubmit;
32   
33       private int currentStatus = OrderStatus.PENDING; // 默认待处理
34   
35       @Override
36       protected void onCreate(Bundle savedInstanceState) {
37           super.onCreate(savedInstanceState);
38           EdgeToEdge.enable(this);
39           setContentView(R.layout.activity_order_create);
40   
41           initViews();
42           setupToolbar();
43           setupStatusDropdown();
44           setupSubmit();
45           prefillFromIntent();
46       }
47   
48       private void initViews() {
49           toolbar = findViewById(R.id.toolbar);
50           etTitle = findViewById(R.id.et_title);
51           etDescription = findViewById(R.id.et_description);
52           spStatus = findViewById(R.id.sp_status);
53           btnSubmit = findViewById(R.id.btn_submit);
54       }
55   
56       private void setupToolbar() {
57           toolbar.setNavigationOnClickListener(new View.OnClickListener() {
58               @Override
59               public void onClick(View v) { finish(); }
60           });
61       }
62   
63       private void setupStatusDropdown() {
64           // 创建工单时只能选择待处理或紧急处理，不能直接创建已完成状态的工单
65           String[] items = new String[]{"待处理", "紧急处理"};
66           ArrayAdapter<String> adapter = new ArrayAdapter<>(this, android.R.layout.simple_list_item_1, items);
67           spStatus.setAdapter(adapter);
68           spStatus.setOnItemClickListener((parent, view, position, id) -> {
69               if (position == 0) {
70                   currentStatus = OrderStatus.PENDING;
71               } else if (position == 1) {
72                   currentStatus = OrderStatus.URGENT;
73               } else {
74                   currentStatus = OrderStatus.PENDING; // 默认待处理
75               }
76           });
77           // 默认选择待处理
78           spStatus.setText(items[0], false);
79           currentStatus = OrderStatus.PENDING;
80       }
81   
82       private void prefillFromIntent() {
83           String hintTitle = getIntent().getStringExtra("hint_title");
84           String hintDesc = getIntent().getStringExtra("hint_desc");
85           if (hintTitle != null) etTitle.setText(hintTitle);
86           if (hintDesc != null) etDescription.setText(hintDesc);
87       }
88   
89       private void setupSubmit() {
90           btnSubmit.setOnClickListener(new View.OnClickListener() {
91               @Override
92               public void onClick(View v) {
93                   String title = etTitle.getText()!=null? etTitle.getText().toString().trim(): "";
94                   String desc = etDescription.getText()!=null? etDescription.getText().toString().trim(): "";
95                   if (title.isEmpty() || desc.isEmpty()) {
96                       Toast.makeText(OrderCreateActivity.this, "请填写标题与描述", Toast.LENGTH_SHORT).show();
97                       return;
98                   }
99                   submitOrder(title, desc, currentStatus);
100              }
101          });
102      }
103  
104      private void submitOrder(String title, String description, int status) {
105          OrderApiService api = RetrofitClient.create(this, OrderApiService.class);
106          CreateOrderRequest request = new CreateOrderRequest(title, description, status);
107          btnSubmit.setEnabled(false);
108          btnSubmit.setText("提交中...");
109          api.createOrder(request).enqueue(new Callback<ApiResponse<Integer>>() {
110              @Override
111              public void onResponse(Call<ApiResponse<Integer>> call, Response<ApiResponse<Integer>> response) {
112                  btnSubmit.setEnabled(true);
113                  btnSubmit.setText("提交");
114                  if (response.isSuccessful() && response.body()!=null && response.body().getCode()==200) {
115                      Toast.makeText(OrderCreateActivity.this, "创建成功，工单#" + response.body().getData(), Toast.LENGTH_LONG).show();
116                      finish();
117                  } else {
118                      String msg = response.body()!=null ? response.body().getMessage() : "创建失败";
119                      Toast.makeText(OrderCreateActivity.this, msg, Toast.LENGTH_LONG).show();
120                  }
121              }
122              @Override
123              public void onFailure(Call<ApiResponse<Integer>> call, Throwable t) {
124                  btnSubmit.setEnabled(true);
125                  btnSubmit.setText("提交");
126                  Toast.makeText(OrderCreateActivity.this, "网络错误: " + t.getMessage(), Toast.LENGTH_LONG).show();
127              }
128          });
129      }
130  }
