1    package com.example.modelfarm;

6    import androidx.appcompat.app.AppCompatActivity;

8    import android.content.DialogInterface;
9    import android.widget.ProgressBar;
10   import android.widget.Toast;
11   import androidx.appcompat.app.AlertDialog;
12   import androidx.recyclerview.widget.LinearLayoutManager;
13   
14   import java.util.List;
15   
16   import com.example.modelfarm.network.RetrofitClient;
17   import com.example.modelfarm.network.models.ApiResponse;
18   import com.example.modelfarm.network.models.UserWitnRole;
19   import com.example.modelfarm.network.services.AdminApiService;
20   import retrofit2.Call;
21   import retrofit2.Callback;
22   import retrofit2.Response;
23   
24   public class AdminActivity extends AppCompatActivity {
25   
26       private RecyclerView recyclerView;
27       private AdminUserAdapter adapter;
28       private ProgressBar progressBar;
29       // 假设你有一个单例类 RetrofitClient 用于获取 ApiService
30   
31       private AdminApiService apiService;
32   
33       @Override
34       protected void onCreate(Bundle savedInstanceState) {
35           super.onCreate(savedInstanceState);
36   
37           // 隐藏默认标题栏
38           if (getSupportActionBar() != null) {
39               getSupportActionBar().hide();
40           }
41   
42           setContentView(R.layout.activity_admin);
43   
44           // 初始化 API (这里假设你已经配置好了 Retrofit)
45           apiService = RetrofitClient.create(this,AdminApiService.class);
46   
47           initView();
48           fetchUserList();
49       }
50   
51       private void initView() {
52           progressBar = findViewById(R.id.progressBar);
53   
54           // 1. 初始化 RecyclerView
55           recyclerView = findViewById(R.id.recyclerView);
56           recyclerView.setLayoutManager(new LinearLayoutManager(this));
57   
58           // 2. 初始化 Adapter 并处理回调
59           adapter = new AdminUserAdapter(new AdminUserAdapter.OnDeleteClickListener() {
60               @Override
61               public void onDeleteClick(UserWitnRole user) {
62                   showDeleteConfirmDialog(user);
63               }
64           });
65           recyclerView.setAdapter(adapter);
66   
67           // 3. 初始化 Header 的返回按钮
68           ImageButton btnBack = findViewById(R.id.btnBack);
69           btnBack.setOnClickListener(new View.OnClickListener() {
70               @Override
71               public void onClick(View v) {
72                   // 关闭当前页面，返回上一页
73                   finish();
74               }
75           });
76       }
77   
78       // 请求用户列表
79       private void fetchUserList() {
80           progressBar.setVisibility(View.VISIBLE);
81   
82           apiService.getAllUsers().enqueue(new Callback<ApiResponse<List<UserWitnRole>>>() {
83               @Override
84               public void onResponse(Call<ApiResponse<List<UserWitnRole>>> call, Response<ApiResponse<List<UserWitnRole>>> response) {
85                   progressBar.setVisibility(View.GONE);
86                   if (response.isSuccessful() && response.body() != null) {
87                       if (response.body().getCode() == 200) {
88                           // 更新列表数据
89                           adapter.setData(response.body().getData());
90                       } else {
91                           Toast.makeText(AdminActivity.this, "获取失败: " + response.body().getMessage(), Toast.LENGTH_SHORT).show();
92                       }
93                   }
94               }
95   
96               @Override
97               public void onFailure(Call<ApiResponse<List<UserWitnRole>>> call, Throwable t) {
98                   progressBar.setVisibility(View.GONE);
99                   Toast.makeText(AdminActivity.this, "网络错误", Toast.LENGTH_SHORT).show();
100              }
101          });
102      }
103  
104      // 弹出删除确认框
105      private void showDeleteConfirmDialog(final UserWitnRole user) {
106          new AlertDialog.Builder(this)
107                  .setTitle("确认移除")
108                  .setMessage("确定要移除用户 " + user.getUser().getUsername() + " 吗？")
109                  .setPositiveButton("确定", new DialogInterface.OnClickListener() {
110                      @Override
111                      public void onClick(DialogInterface dialog, int which) {
112                          deleteUserFromApi(user);
113                      }
114                  })
115                  .setNegativeButton("取消", null)
116                  .show();
117      }
118  
119      // 调用 API 删除用户
120      private void deleteUserFromApi(final UserWitnRole user) {
121          progressBar.setVisibility(View.VISIBLE);
122  
123          apiService.deleteUser(user.getUser().getId()).enqueue(new Callback<ApiResponse<Object>>() {
124              @Override
125              public void onResponse(Call<ApiResponse<Object>> call, Response<ApiResponse<Object>> response) {
126                  progressBar.setVisibility(View.GONE);
127                  if (response.isSuccessful() && response.body() != null) {
128                      if (response.body().getCode() == 200) {
129                          Toast.makeText(AdminActivity.this, "删除成功", Toast.LENGTH_SHORT).show();
130                          // 从界面移除该项，不用重新请求整个列表
131                          adapter.removeItem(user);
132                      } else {
133                          Toast.makeText(AdminActivity.this, "删除失败: " + response.body().getMessage(), Toast.LENGTH_SHORT).show();
134                      }
135                  }
136              }
137  
138              @Override
139              public void onFailure(Call<ApiResponse<Object>> call, Throwable t) {
140                  progressBar.setVisibility(View.GONE);
141                  Toast.makeText(AdminActivity.this, "网络错误", Toast.LENGTH_SHORT).show();
142              }
143          });
144      }
145  }
