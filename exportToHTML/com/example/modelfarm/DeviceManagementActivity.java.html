113      private void initBottomNavigation() {
114          BottomNavigationView bottomNavigationView = findViewById(R.id.bottomNavigation);
115          if (bottomNavigationView == null) {
116              return;
117          }
118  
119          bottomNavigationView.setSelectedItemId(R.id.menu_devices);
120          bottomNavigationView.setOnItemSelectedListener(item -> {
121              int itemId = item.getItemId();
122              if (itemId == R.id.menu_devices) {
123                  return true;
124              } else if (itemId == R.id.menu_dashboard) {
125                  startActivity(new Intent(DeviceManagementActivity.this, DashboardActivity.class));
126              } else if (itemId == R.id.menu_farms) {
127                  startActivity(new Intent(DeviceManagementActivity.this, farm_list.class));
128              } else if (itemId == R.id.menu_orders) {
129                  startActivity(new Intent(DeviceManagementActivity.this, OrderListActivity.class));
130              } else if (itemId == R.id.menu_company) {
131                  startActivity(new Intent(DeviceManagementActivity.this, company_info.class));
132              } else {
133                  return false;
134              }
135              finish();
136              overridePendingTransition(R.anim.slide_in_right, R.anim.slide_out_left);
137              return true;
138          });
139      }

162      private void openLiveStream(Device device) {
163          String pushName = device.getPushName();
164          if (TextUtils.isEmpty(pushName)) {
165              Toast.makeText(this, "该摄像头缺少推流名称，无法打开直播", Toast.LENGTH_LONG).show();
166              return;
167          }
168  
169          Intent intent = new Intent(this, LiveStreamActivity.class);
170          intent.putExtra("device_id", device.getId());
171          intent.putExtra("device_name", device.getName());
172          intent.putExtra("push_name", pushName);
173          if (!TextUtils.isEmpty(device.getUrl())) {
174              intent.putExtra("stream_url", device.getUrl());
175          }
176          startActivity(intent);
177      }

195      private void loadDeviceData() {
196          deviceApi.getDeviceList(1, 100, null, null, null, null).enqueue(new retrofit2.Callback<com.example.modelfarm.network.models.ApiResponse<com.example.modelfarm.network.models.PageResponse<Device>>>() {
197              @Override
198              public void onResponse(retrofit2.Call<com.example.modelfarm.network.models.ApiResponse<com.example.modelfarm.network.models.PageResponse<Device>>> call, retrofit2.Response<com.example.modelfarm.network.models.ApiResponse<com.example.modelfarm.network.models.PageResponse<Device>>> response) {
199                  runOnUiThread(() -> {
200                      deviceList.clear();
201                      if (response.isSuccessful() && response.body() != null && response.body().getCode() == 200) {
202                          List<Device> apiList = response.body().getData().getRecords();
203                          deviceList.addAll(apiList);
204                      } else {
205                          Toast.makeText(DeviceManagementActivity.this, "获取设备数据失败: " + (response.body()!=null?response.body().getMessage():"接口异常"), Toast.LENGTH_LONG).show();
206                      }
207                      deviceAdapter.notifyDataSetChanged();
208                      updateDeviceStatistics();
209                  });
210              }
211              @Override
212              public void onFailure(retrofit2.Call<com.example.modelfarm.network.models.ApiResponse<com.example.modelfarm.network.models.PageResponse<Device>>> call, Throwable t) {
213                  runOnUiThread(() -> {
214                      Toast.makeText(DeviceManagementActivity.this, "获取设备数据失败: 网络异常", Toast.LENGTH_LONG).show();
215                      deviceList.clear();
216                      deviceAdapter.notifyDataSetChanged();
217                      updateDeviceStatistics();
218                  });
219              }
220          });
221      }

223      /**
224       * 更新设备统计信息
225       */
226      private void updateDeviceStatistics() {
227          int total = deviceList.size();
228          int online = 0;
229          int offline = 0;
230  
231          for (Device device : deviceList) {
232              if ("在线".equals(device.getStatus())) {
233                  online++;
234              } else {
235                  offline++;
236              }
237          }
238  
239          tvTotalDevices.setText(String.valueOf(total));
240          tvOnlineDevices.setText(String.valueOf(online));
241          tvOfflineDevices.setText(String.valueOf(offline));
242      }

342       * 加载所有农场的站点列表
343       */
344      private void loadAllFarmSites(Runnable onComplete) {
345          // 先获取所有农场
346          farmApiService.getEnterpriseFarms().enqueue(new Callback<ApiResponse<java.util.List<com.example.modelfarm.network.models.Farm>>>() {
347              @Override
348              public void onResponse(Call<ApiResponse<java.util.List<com.example.modelfarm.network.models.Farm>>> call, Response<ApiResponse<java.util.List<com.example.modelfarm.network.models.Farm>>> response) {
349                  if (response.isSuccessful() && response.body() != null && response.body().getCode() == 200) {
350                      java.util.List<com.example.modelfarm.network.models.Farm> farms = response.body().getData();
351                      farmSiteList.clear();
352                      
353                      // 为每个农场加载站点
354                      final int[] completed = {0};
355                      final int total = farms.size();
356                      
357                      if (total == 0) {
358                          runOnUiThread(onComplete);
359                          return;
360                      }
361                      
362                      for (com.example.modelfarm.network.models.Farm farm : farms) {
363                          farmApiService.getFarmSites(farm.getId()).enqueue(new Callback<ApiResponse<java.util.List<FarmSite>>>() {
364                              @Override
365                              public void onResponse(Call<ApiResponse<java.util.List<FarmSite>>> call, Response<ApiResponse<java.util.List<FarmSite>>> response) {
366                                  synchronized (farmSiteList) {
367                                      if (response.isSuccessful() && response.body() != null && response.body().getCode() == 200) {
368                                          java.util.List<FarmSite> sites = response.body().getData();
369                                          if (sites != null) {
370                                              farmSiteList.addAll(sites);
371                                          }
372                                      }
373                                      completed[0]++;
374                                      if (completed[0] == total) {
375                                          runOnUiThread(onComplete);
376                                      }
377                                  }
378                              }
379  
380                              @Override
381                              public void onFailure(Call<ApiResponse<java.util.List<FarmSite>>> call, Throwable t) {
382                                  synchronized (farmSiteList) {
383                                      completed[0]++;
384                                      if (completed[0] == total) {
385                                          runOnUiThread(onComplete);
386                                      }
387                                  }
388                              }
389                          });
390                      }
391                  } else {
392                      runOnUiThread(onComplete);
393                  }
394              }
395  
396              @Override
397              public void onFailure(Call<ApiResponse<java.util.List<com.example.modelfarm.network.models.Farm>>> call, Throwable t) {
398                  runOnUiThread(onComplete);
399              }
400          });
401      }
