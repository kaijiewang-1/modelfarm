AuthManager.kt 
1    package com.example.modelfarm.network
2    
3    import android.content.Context
4    import android.content.SharedPreferences
5    import com.example.modelfarm.network.models.LoginResponse
6    import com.example.modelfarm.network.models.SaToken
7    
8    /**
9     * 认证管理器
10    * 负责管理用户登录状态和token
11    */
12   class AuthManager private constructor(context: Context) {
13       
14       companion object {
15           private const val PREFS_NAME = "SmartFarmAuth"
16           private const val KEY_TOKEN = "auth_token"
17           private const val KEY_USER_ID = "user_id"
18           private const val KEY_ENTERPRISE_ID = "enterprise_id"
19           private const val KEY_IS_LOGGED_IN = "is_logged_in"
20           
21           @Volatile
22           private var INSTANCE: AuthManager? = null
23           
24           @JvmStatic
25           fun getInstance(context: Context): AuthManager {
26               return INSTANCE ?: synchronized(this) {
27                   INSTANCE ?: AuthManager(context.applicationContext).also { INSTANCE = it }
28               }
29           }
30       }
31       
32       private val prefs: SharedPreferences = context.getSharedPreferences(PREFS_NAME, Context.MODE_PRIVATE)
33       
34       /**
35        * 保存登录信息
36        */
37       fun saveLoginInfo(loginResponse: LoginResponse) {
38           prefs.edit().apply {
39               putString(KEY_TOKEN, loginResponse.satoken.tokenValue)
40               putInt(KEY_USER_ID, loginResponse.userId)
41               putInt(KEY_ENTERPRISE_ID, loginResponse.enterpriseId)
42               putBoolean(KEY_IS_LOGGED_IN, true)
43               apply()
44           }
45           
46           // 保存认证token到SharedPreferences
47           // RetrofitClient会自动从AuthManager获取token
48       }
49       
50       /**
51        * 获取当前token
52        */
53       fun getToken(): String? {
54           return prefs.getString(KEY_TOKEN, null)
55       }
56       
57       /**
58        * 获取当前用户ID
59        */
60       fun getUserId(): Int {
61           return prefs.getInt(KEY_USER_ID, -1)
62       }
63       
64       /**
65        * 获取当前企业ID
66        */
67       fun getEnterpriseId(): Int {
68           return prefs.getInt(KEY_ENTERPRISE_ID, -1)
69       }
70       
71       /**
72        * 检查是否已登录
73        */
74       fun isLoggedIn(): Boolean {
75           return prefs.getBoolean(KEY_IS_LOGGED_IN, false) && getToken() != null
76       }
77       
78       /**
79        * 登出
80        */
81       fun logout() {
82           prefs.edit().clear().apply()
83           // RetrofitClient会自动从AuthManager获取token
84       }
85       
86       /**
87        * 更新token
88        */
89       fun updateToken(token: String) {
90           prefs.edit().putString(KEY_TOKEN, token).apply()
91           // RetrofitClient会自动从AuthManager获取token
92       }
93   }
