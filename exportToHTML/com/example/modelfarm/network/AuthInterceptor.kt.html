1    package com.example.modelfarm.network
2    
3    import android.content.Context
4    import okhttp3.Interceptor
5    import okhttp3.Response
6    
7    /**
8     * 认证拦截器
9     * 自动添加satoken到请求头（从AuthManager动态获取）
10    */
11   class AuthInterceptor(private val context: Context) : Interceptor {
12       override fun intercept(chain: Interceptor.Chain): Response {
13           val originalRequest = chain.request()
14           val token = AuthManager.getInstance(context).getToken()
15           val newRequest = if (!token.isNullOrEmpty()) {
16               originalRequest.newBuilder()
17                   .addHeader("satoken", token)
18                   .build()
19           } else {
20               originalRequest
21           }
22           return chain.proceed(newRequest)
23       }
24   }
