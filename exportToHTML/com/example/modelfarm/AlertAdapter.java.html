1    package com.example.modelfarm;
2    
3    import android.view.LayoutInflater;
4    import android.view.View;
5    import android.view.ViewGroup;
6    import android.widget.TextView;
7    
8    import androidx.annotation.NonNull;
9    import androidx.recyclerview.widget.RecyclerView;
10   
11   import com.example.modelfarm.R;
12   import com.google.android.material.card.MaterialCardView;
13   
14   import java.util.List;
15   
16   public class AlertAdapter extends RecyclerView.Adapter<AlertAdapter.AlertViewHolder> {
17   
18       private final List<EnvironmentMonitoringEnhancedActivity.Alert> alertList;
19   
20       public AlertAdapter(List<EnvironmentMonitoringEnhancedActivity.Alert> alertList) {
21           this.alertList = alertList;
22       }
23   
24       @NonNull
25       @Override
26       public AlertViewHolder onCreateViewHolder(@NonNull ViewGroup parent, int viewType) {
27           View view = LayoutInflater.from(parent.getContext())
28                   .inflate(R.layout.item_alert, parent, false);
29           return new AlertViewHolder(view);
30       }
31   
32       @Override
33       public void onBindViewHolder(@NonNull AlertViewHolder holder, int position) {
34           EnvironmentMonitoringEnhancedActivity.Alert alert = alertList.get(position);
35           holder.bind(alert);
36       }
37   
38       @Override
39       public int getItemCount() {
40           return alertList.size();
41       }
42   
43       class AlertViewHolder extends RecyclerView.ViewHolder {
44           private final MaterialCardView cardAlert;
45           private final TextView tvAlertTitle;
46           private final TextView tvAlertMessage;
47           private final TextView tvAlertTime;
48           private final TextView tvAlertLevel;
49   
50           public AlertViewHolder(@NonNull View itemView) {
51               super(itemView);
52               cardAlert = itemView.findViewById(R.id.card_alert);
53               tvAlertTitle = itemView.findViewById(R.id.tv_alert_title);
54               tvAlertMessage = itemView.findViewById(R.id.tv_alert_message);
55               tvAlertTime = itemView.findViewById(R.id.tv_alert_time);
56               tvAlertLevel = itemView.findViewById(R.id.tv_alert_level);
57           }
58   
59           public void bind(EnvironmentMonitoringEnhancedActivity.Alert alert) {
60               tvAlertTitle.setText(alert.getTitle());
61               tvAlertMessage.setText(alert.getMessage());
62               tvAlertTime.setText(alert.getTime());
63               tvAlertLevel.setText(alert.getLevel().toUpperCase());
64   
65               // 根据报警级别设置颜色
66               switch (alert.getLevel()) {
67                   case "high":
68                       tvAlertLevel.setTextColor(0xFFF44336);
69                       cardAlert.setStrokeColor(0xFFF44336);
70                       break;
71                   case "medium":
72                       tvAlertLevel.setTextColor(0xFFFF9800);
73                       cardAlert.setStrokeColor(0xFFFF9800);
74                       break;
75                   case "low":
76                       tvAlertLevel.setTextColor(0xFF4CAF50);
77                       cardAlert.setStrokeColor(0xFF4CAF50);
78                       break;
79               }
80           }
81       }
82   }
