UserDataDisplay.java 
1    package com.example.modelfarm.utils;
2    
3    import com.example.modelfarm.network.models.User;
4    
5    /**
6     * 用户数据展示器
7     * 用于处理用户信息的显示逻辑
8     */
9    public class UserDataDisplay {
10       
11       /**
12        * 获取用户显示名称
13        */
14       public static String getUserDisplayName(User user) {
15           if (!user.getUsername().isEmpty()) {
16               return user.getUsername();
17           }
18           
19           if (!user.getPhone().isEmpty()) {
20               return user.getPhone();
21           }
22           
23           return "未知用户";
24       }
25       
26       /**
27        * 获取用户状态显示
28        */
29       public static String getUserStatusDisplay(User user) {
30           String status = user.getStatus();
31           if (status == null) return "未知";
32           String s = status.trim().toLowerCase();
33           if ("1".equals(s) || "active".equals(s) || "enabled".equals(s) || "正常".equals(s)) {
34               return "活跃";
35           }
36           if ("0".equals(s) || "inactive".equals(s) || "disabled".equals(s) || "禁用".equals(s)) {
37               return "非活跃";
38           }
39           if ("-1".equals(s) || "paused".equals(s) || "suspended".equals(s)) {
40               return "已暂停";
41           }
42           return status;
43       }
44       
45       /**
46        * 获取用户角色显示
47        */
48       public static String getUserRoleDisplay(User user) {
49           // 根据用户ID或企业ID判断角色
50           if (user.getEnterpriseId() > 0) {
51               return "企业用户";
52           }
53           
54           return "普通用户";
55       }
56       
57       /**
58        * 获取用户创建时间显示
59        */
60       public static String getUserCreateTimeDisplay(User user) {
61           if (user.getCreatedAt() != null && !user.getCreatedAt().isEmpty()) {
62               return user.getCreatedAt();
63           }
64           
65           return "未知时间";
66       }
67       
68       /**
69        * 获取用户最后更新时间显示
70        */
71       public static String getUserUpdateTimeDisplay(User user) {
72           if (user.getUpdatedAt() != null && !user.getUpdatedAt().isEmpty()) {
73               return user.getUpdatedAt();
74           }
75           
76           return "未知时间";
77       }
78       
79       /**
80        * 获取用户完整信息显示
81        */
82       public static String getUserFullInfoDisplay(User user) {
83           StringBuilder info = new StringBuilder();
84           info.append("用户名: ").append(getUserDisplayName(user)).append("\n");
85           info.append("手机号: ").append(user.getPhone() != null ? user.getPhone() : "未知").append("\n");
86           info.append("状态: ").append(getUserStatusDisplay(user)).append("\n");
87           info.append("角色: ").append(getUserRoleDisplay(user)).append("\n");
88           info.append("创建时间: ").append(getUserCreateTimeDisplay(user)).append("\n");
89           return info.toString();
90       }
91       
92       /**
93        * 检查用户是否有效
94        */
95       public static boolean isValidUser(User user) {
96           return user != null && 
97                  user.getUsername() != null && 
98                  !user.getUsername().isEmpty() &&
99                  user.getPhone() != null && 
100                 !user.getPhone().isEmpty();
101      }
102      
103      /**
104       * 获取用户权限级别
105       */
106      public static String getUserPermissionLevel(User user) {
107          if (user.getEnterpriseId() > 0) {
108              return "企业管理员";
109          }
110          
111          return "普通用户";
112      }
113  }
