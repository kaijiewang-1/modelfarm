171      @Override
172      protected void onActivityResult(int requestCode, int resultCode, Intent data) {
173          super.onActivityResult(requestCode, resultCode, data);
174          if (resultCode == RESULT_OK && data != null) {
175              Uri imageUri = data.getData();
176              if (imageUri != null) {
177                  if (requestCode == REQUEST_CODE_PICK_SOLUTION_IMAGE) {
178                      solutionImageUris.add(imageUri);
179                      addImageToLayout(imageUri, llSolutionImages, true);
180                  } else if (requestCode == REQUEST_CODE_PICK_INSPECTION_IMAGE) {
181                      inspectionImageUris.add(imageUri);
182                      addImageToLayout(imageUri, llInspectionImages, false);
183                  }
184              }
185          }
186      }

191      private void addImageToLayout(Uri imageUri, LinearLayout container, boolean isSolution) {
192          try {
193              // 创建图片视图
194              ImageView imageView = new ImageView(this);
195              LinearLayout.LayoutParams params = new LinearLayout.LayoutParams(
196                  (int) (120 * getResources().getDisplayMetrics().density),
197                  (int) (120 * getResources().getDisplayMetrics().density)
198              );
199              params.setMargins(0, 0, (int) (8 * getResources().getDisplayMetrics().density), 0);
200              imageView.setLayoutParams(params);
201              imageView.setScaleType(ImageView.ScaleType.CENTER_CROP);
202              
203              // 加载图片
204              InputStream inputStream = getContentResolver().openInputStream(imageUri);
205              Bitmap bitmap = BitmapFactory.decodeStream(inputStream);
206              imageView.setImageBitmap(bitmap);
207              if (inputStream != null) inputStream.close();
208              
209              // 添加删除按钮
210              ImageView deleteBtn = new ImageView(this);
211              LinearLayout.LayoutParams deleteParams = new LinearLayout.LayoutParams(
212                  (int) (24 * getResources().getDisplayMetrics().density),
213                  (int) (24 * getResources().getDisplayMetrics().density)
214              );
215              deleteBtn.setLayoutParams(deleteParams);
216              deleteBtn.setImageResource(android.R.drawable.ic_menu_delete);
217              deleteBtn.setBackgroundColor(0x80FF0000);
218              deleteBtn.setPadding(4, 4, 4, 4);
219              
220              // 创建容器
221              LinearLayout imageContainer = new LinearLayout(this);
222              imageContainer.setOrientation(LinearLayout.VERTICAL);
223              LinearLayout.LayoutParams containerParams = new LinearLayout.LayoutParams(
224                  LinearLayout.LayoutParams.WRAP_CONTENT,
225                  LinearLayout.LayoutParams.WRAP_CONTENT
226              );
227              containerParams.setMargins(0, 0, (int) (8 * getResources().getDisplayMetrics().density), 0);
228              imageContainer.setLayoutParams(containerParams);
229              
230              imageContainer.addView(imageView);
231              imageContainer.addView(deleteBtn);
232              
233              // 删除按钮点击事件
234              final Uri uriToRemove = imageUri;
235              deleteBtn.setOnClickListener(new View.OnClickListener() {
236                  @Override
237                  public void onClick(View v) {
238                      if (isSolution) {
239                          solutionImageUris.remove(uriToRemove);
240                      } else {
241                          inspectionImageUris.remove(uriToRemove);
242                      }
243                      container.removeView(imageContainer);
244                  }
245              });
246              
247              container.addView(imageContainer);
248          } catch (IOException e) {
249              Toast.makeText(this, "加载图片失败: " + e.getMessage(), Toast.LENGTH_SHORT).show();
250          }
251      }

262      private void loadOrderDetail() {
263          OrderApiService api = RetrofitClient.create(this, OrderApiService.class);
264          api.getOrder(orderId).enqueue(new Callback<ApiResponse<Order>>() {
265              @Override
266              public void onResponse(Call<ApiResponse<Order>> call, Response<ApiResponse<Order>> response) {
267                  if (response.isSuccessful() && response.body() != null && response.body().getCode() == 200 && response.body().getData() != null) {
268                      currentOrder = response.body().getData();
269                      displayOrderDetail(currentOrder);
270                  } else {
271                      String msg = response.body() != null ? response.body().getMessage() : "获取工单详情失败";
272                      Toast.makeText(OrderDetailActivity.this, msg, Toast.LENGTH_LONG).show();
273                      finish();
274                  }
275              }
276  
277              @Override
278              public void onFailure(Call<ApiResponse<Order>> call, Throwable t) {
279                  Toast.makeText(OrderDetailActivity.this, "网络错误: " + t.getMessage(), Toast.LENGTH_LONG).show();
280                  finish();
281              }
282          });
283      }

285      private void displayOrderDetail(Order order) {
286          tvOrderId.setText("工单编号: #" + order.getId());
287          tvTitle.setText(order.getTitle());
288          tvDescription.setText(order.getDescription());
289          tvStatus.setText(getStatusText(order.getStatus()));
290          tvCreatorId.setText("创建人ID: " + order.getCreatorId());
291  
292          if (order.getAcceptedId() != null) {
293              tvAcceptedId.setText("受理人ID: " + order.getAcceptedId());
294          } else {
295              tvAcceptedId.setText("受理人: 未派发");
296          }
297  
298          if (order.getSolvedId() != null) {
299              tvSolvedId.setText("处理人ID: " + order.getSolvedId());
300          } else {
301              tvSolvedId.setText("处理人: 未处理");
302          }
303  
304          tvCreatedAt.setText("创建时间: " + order.getCreatedAt());
305  
306          if (order.getCompletedAt() != null && !order.getCompletedAt().isEmpty()) {
307              tvCompletedAt.setText("完成时间: " + order.getCompletedAt());
308          } else {
309              tvCompletedAt.setText("完成时间: 未完成");
310          }
311  
312          tvUpdatedAt.setText("更新时间: " + order.getUpdatedAt());
313  
314          // 显示解决方案
315          if (order.getSolution() != null && !order.getSolution().isEmpty()) {
316              etSolution.setText(order.getSolution());
317          }
318          
319          // 显示解决图片
320          if (order.getSolutionImages() != null && !order.getSolutionImages().isEmpty()) {
321              loadImageUrls(order.getSolutionImages(), solutionImageUrls);
322              displayImages(solutionImageUrls, llSolutionImages, true);
323          }
324          
325          // 显示异常情况和巡查图片（仅普通工单和紧急工单）
326          int status = order.getStatus();
327          boolean showInspection = (status == OrderStatus.PENDING || status == OrderStatus.URGENT);
328          
329          if (showInspection) {
330              cardInspection.setVisibility(View.VISIBLE);
331              
332              if (order.getInspectionIssue() != null && !order.getInspectionIssue().isEmpty()) {
333                  etInspectionIssue.setText(order.getInspectionIssue());
334              }
335              
336              if (order.getInspectionImages() != null && !order.getInspectionImages().isEmpty()) {
337                  loadImageUrls(order.getInspectionImages(), inspectionImageUrls);
338                  displayImages(inspectionImageUrls, llInspectionImages, false);
339              }
340          } else {
341              cardInspection.setVisibility(View.GONE);
342          }
343  
344          // 根据接口文档：只能标记待处理(1)或紧急处理(3)状态的工单
345          // 不能标记已完成(2)状态的工单
346          boolean canComplete = (status == OrderStatus.PENDING || status == OrderStatus.URGENT);
347  
348          if (canComplete) {
349              btnComplete.setVisibility(View.VISIBLE);
350          } else {
351              btnComplete.setVisibility(View.GONE);
352          }
353      }

355      /**
356       * 加载图片URL列表
357       */
358      private void loadImageUrls(String imageUrlsJson, List<String> targetList) {
359          targetList.clear();
360          if (imageUrlsJson == null || imageUrlsJson.isEmpty()) {
361              return;
362          }
363          
364          try {
365              // 尝试解析JSON数组
366              Gson gson = new Gson();
367              List<String> urls = gson.fromJson(imageUrlsJson, new TypeToken<List<String>>(){}.getType());
368              if (urls != null) {
369                  targetList.addAll(urls);
370              }
371          } catch (Exception e) {
372              // 如果不是JSON，尝试按逗号分隔
373              String[] urls = imageUrlsJson.split(",");
374              for (String url : urls) {
375                  String trimmed = url.trim();
376                  if (!trimmed.isEmpty()) {
377                      targetList.add(trimmed);
378                  }
379              }
380          }
381      }

406       * 保存工单更改

408      private void saveOrderChanges() {
409          if (currentOrder == null) {
410              Toast.makeText(this, "工单数据未加载", Toast.LENGTH_SHORT).show();
411              return;
412          }
413          
414          String solution = etSolution.getText() != null ? etSolution.getText().toString().trim() : "";
415          String inspectionIssue = etInspectionIssue.getText() != null ? etInspectionIssue.getText().toString().trim() : "";
416          
417          // 合并图片URLs（新上传的图片需要先上传到服务器获取URL，这里简化处理）
418          // 实际应该先上传图片，获取URL后再保存
419          String solutionImagesJson = buildImageUrlsJson(solutionImageUrls, solutionImageUris);
420          String inspectionImagesJson = buildImageUrlsJson(inspectionImageUrls, inspectionImageUris);
421          
422          UpdateOrderRequest request = new UpdateOrderRequest(
423              currentOrder.getId(),
424              null, // title
425              null, // description
426              null, // status
427              solution.isEmpty() ? null : solution,
428              solutionImagesJson.isEmpty() ? null : solutionImagesJson,
429              inspectionIssue.isEmpty() ? null : inspectionIssue,
430              inspectionImagesJson.isEmpty() ? null : inspectionImagesJson
431          );
432          
433          OrderApiService api = RetrofitClient.create(this, OrderApiService.class);
434          api.updateOrder(request).enqueue(new Callback<ApiResponse<Void>>() {
435              @Override
436              public void onResponse(Call<ApiResponse<Void>> call, Response<ApiResponse<Void>> response) {
437                  if (response.isSuccessful() && response.body() != null && response.body().getCode() == 200) {
438                      Toast.makeText(OrderDetailActivity.this, "保存成功", Toast.LENGTH_SHORT).show();
439                      // 重新加载工单详情
440                      loadOrderDetail();
441                  } else {
442                      String msg = response.body() != null ? response.body().getMessage() : "保存失败";
443                      Toast.makeText(OrderDetailActivity.this, msg, Toast.LENGTH_LONG).show();
444                  }
445              }
446  
447              @Override
448              public void onFailure(Call<ApiResponse<Void>> call, Throwable t) {
449                  Toast.makeText(OrderDetailActivity.this, "网络错误: " + t.getMessage(), Toast.LENGTH_LONG).show();
450              }
451          });
452      }

475      private String getStatusText(int status) {
476          switch (status) {
477              case OrderStatus.PENDING:
478                  return "待处理";
479              case OrderStatus.COMPLETED:
480                  return "已完成";
481              case OrderStatus.URGENT:
482                  return "紧急处理";
483              case OrderStatus.CLAIMED:
484                  return "受理中";
485              default:
486                  return String.valueOf(status);
487          }
488      }

490      private void handleCompleteOrder(Order order) {
491          OrderApiService api = RetrofitClient.create(this, OrderApiService.class);
492          api.completeOrder(order.getId()).enqueue(new Callback<ApiResponse<Void>>() {
493              @Override
494              public void onResponse(Call<ApiResponse<Void>> call, Response<ApiResponse<Void>> response) {
495                  if (response.isSuccessful() && response.body() != null && response.body().getCode() == 200) {
496                      Toast.makeText(OrderDetailActivity.this, "工单标记为已完成", Toast.LENGTH_SHORT).show();
497                      // 重新加载工单详情
498                      loadOrderDetail();
499                  } else {
500                      String msg = response.body() != null ? response.body().getMessage() : "完成失败";
501                      Toast.makeText(OrderDetailActivity.this, msg, Toast.LENGTH_LONG).show();
502                  }
503              }
504  
505              @Override
506              public void onFailure(Call<ApiResponse<Void>> call, Throwable t) {
507                  Toast.makeText(OrderDetailActivity.this, "网络错误: " + t.getMessage(), Toast.LENGTH_LONG).show();
508              }
