14   import java.util.List;

20   public class AdminUserAdapter extends RecyclerView.Adapter<AdminUserAdapter.UserViewHolder> {
21   
22       private List<UserWitnRole> userList = new ArrayList<>();
23       private OnDeleteClickListener deleteListener;
24   
25       // 定义点击事件接口
26       public interface OnDeleteClickListener {
27           void onDeleteClick(UserWitnRole user);
28       }
29   
30       public AdminUserAdapter(OnDeleteClickListener listener) {
31           this.deleteListener = listener;
32       }
33   
34       // 设置新数据
35       public void setData(List<UserWitnRole> newList) {
36           this.userList = newList;
37           notifyDataSetChanged();
38       }
39   
40       // 删除单项数据并刷新（优化体验，不需要全量刷新）
41       public void removeItem(UserWitnRole item) {
42           int position = userList.indexOf(item);
43           if (position != -1) {
44               userList.remove(position);
45               notifyItemRemoved(position);
46           }
47       }
48   
49       @NonNull
50       @Override
51       public UserViewHolder onCreateViewHolder(@NonNull ViewGroup parent, int viewType) {
52           View view = LayoutInflater.from(parent.getContext())
53                   .inflate(R.layout.item_admin_user, parent, false);
54           return new UserViewHolder(view);
55       }
56   
57       @Override
58       public void onBindViewHolder(@NonNull UserViewHolder holder, int position) {
59           UserWitnRole item = userList.get(position);
60   
61           holder.tvUsername.setText(item.getUser().getUsername());
62           holder.tvRole.setText(item.getRole());
63   
64           // 绑定删除按钮事件
65           holder.btnDelete.setOnClickListener(v -> {
66               if (deleteListener != null) {
67                   deleteListener.onDeleteClick(item);
68               }
69           });
70       }
71   
72       @Override
73       public int getItemCount() {
74           return userList.size();
75       }
76   
77       // ViewHolder 内部类
78       static class UserViewHolder extends RecyclerView.ViewHolder {
79           TextView tvUsername, tvRole;
80           Button btnDelete;
81   
82           public UserViewHolder(@NonNull View itemView) {
83               super(itemView);
84               tvUsername = itemView.findViewById(R.id.tvUsername);
85               tvRole = itemView.findViewById(R.id.tvRole);
86               btnDelete = itemView.findViewById(R.id.btnDelete);
87           }
88       }
