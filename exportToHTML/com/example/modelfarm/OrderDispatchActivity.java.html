1    package com.example.modelfarm;
2    
3    import android.os.Bundle;
4    import android.view.View;
5    import android.widget.ArrayAdapter;
6    import android.widget.AutoCompleteTextView;
7    import android.widget.ImageButton;
8    import android.widget.TextView;
9    import android.widget.Toast;
10   
11   import androidx.activity.EdgeToEdge;
12   import androidx.appcompat.app.AppCompatActivity;
13   
14   import com.example.modelfarm.network.RetrofitClient;
15   import com.example.modelfarm.network.models.ApiResponse;
16   import com.example.modelfarm.network.models.DispatchOrderRequest;
17   import com.example.modelfarm.network.models.EnterpriseUser;
18   import com.example.modelfarm.network.services.EnterpriseApiService;
19   import com.example.modelfarm.network.services.OrderApiService;
20   import com.google.android.material.button.MaterialButton;
21   
22   import java.util.ArrayList;
23   import java.util.List;
24   
25   import retrofit2.Call;
26   import retrofit2.Callback;
27   import retrofit2.Response;
28   
29   /**
30    * 工单派发页面
31    * 使用 POST /order/dispatch 派发工单给指定用户
32    */
33   public class OrderDispatchActivity extends AppCompatActivity {
34   
35       private ImageButton btnBack;
36       private TextView tvTitle;
37       private AutoCompleteTextView spUsers;
38       private MaterialButton btnDispatch;
39       private TextView tvOrderInfo;
40   
41       private int orderId;
42       private String orderTitle;
43       private List<EnterpriseUser> userList = new ArrayList<>();
44       private int selectedUserId = -1;
45   
46       @Override
47       protected void onCreate(Bundle savedInstanceState) {
48           super.onCreate(savedInstanceState);
49           EdgeToEdge.enable(this);
50           setContentView(R.layout.activity_order_dispatch);
51   
52           // 获取传递的工单信息
53           orderId = getIntent().getIntExtra("orderId", -1);
54           orderTitle = getIntent().getStringExtra("orderTitle");
55   
56           if (orderId == -1 || orderTitle == null) {
57               Toast.makeText(this, "工单信息无效", Toast.LENGTH_SHORT).show();
58               finish();
59               return;
60           }
61   
62           initViews();
63           setupClicks();
64           loadEnterpriseUsers();
65       }
66   
67       private void initViews() {
68           btnBack = findViewById(R.id.btnBack);
69           tvTitle = findViewById(R.id.tvTitle);
70           spUsers = findViewById(R.id.spUsers);
71           btnDispatch = findViewById(R.id.btnDispatch);
72           tvOrderInfo = findViewById(R.id.tvOrderInfo);
73   
74           tvTitle.setText("派发工单");
75           tvOrderInfo.setText("工单: " + orderTitle);
76       }
77   
78       private void setupClicks() {
79           btnBack.setOnClickListener(new View.OnClickListener() {
80               @Override
81               public void onClick(View v) {
82                   finish();
83               }
84           });
85   
86           btnDispatch.setOnClickListener(new View.OnClickListener() {
87               @Override
88               public void onClick(View v) {
89                   if (selectedUserId == -1) {
90                       Toast.makeText(OrderDispatchActivity.this, "请选择派发人员", Toast.LENGTH_SHORT).show();
91                       return;
92                   }
93                   dispatchOrder();
94               }
95           });
96       }
97   
98       private void loadEnterpriseUsers() {
99           EnterpriseApiService api = RetrofitClient.create(this, EnterpriseApiService.class);
100          // 获取当前企业用户列表
101          api.getEnterpriseUsers().enqueue(new Callback<ApiResponse<List<EnterpriseUser>>>() {
102              @Override
103              public void onResponse(Call<ApiResponse<List<EnterpriseUser>>> call, Response<ApiResponse<List<EnterpriseUser>>> response) {
104                  if (response.isSuccessful() && response.body() != null && response.body().getCode() == 200 && response.body().getData() != null) {
105                      userList.clear();
106                      userList.addAll(response.body().getData());
107                      setupUserDropdown();
108                  } else {
109                      String msg = response.body() != null ? response.body().getMessage() : "获取用户列表失败";
110                      Toast.makeText(OrderDispatchActivity.this, msg, Toast.LENGTH_LONG).show();
111                  }
112              }
113  
114              @Override
115              public void onFailure(Call<ApiResponse<List<EnterpriseUser>>> call, Throwable t) {
116                  Toast.makeText(OrderDispatchActivity.this, "网络错误:" + t.getMessage(), Toast.LENGTH_LONG).show();
117              }
118          });
119      }
120  
121      private void setupUserDropdown() {
122          List<String> userNames = new ArrayList<>();
123          for (EnterpriseUser user : userList) {
124              userNames.add(user.getUsername() + " (ID: " + user.getId() + ")");
125          }
126  
127          ArrayAdapter<String> adapter = new ArrayAdapter<>(this, android.R.layout.simple_list_item_1, userNames);
128          spUsers.setAdapter(adapter);
129  
130          spUsers.setOnItemClickListener((parent, view, position, id) -> {
131              selectedUserId = userList.get(position).getId();
132          });
133      }
134  
135      private void dispatchOrder() {
136          OrderApiService api = RetrofitClient.create(this, OrderApiService.class);
137          DispatchOrderRequest request = new DispatchOrderRequest(orderId, selectedUserId);
138  
139          btnDispatch.setEnabled(false);
140          btnDispatch.setText("派发中...");
141  
142          api.dispatchOrder(request).enqueue(new Callback<ApiResponse<Void>>() {
143              @Override
144              public void onResponse(Call<ApiResponse<Void>> call, Response<ApiResponse<Void>> response) {
145                  btnDispatch.setEnabled(true);
146                  btnDispatch.setText("派发");
147  
148                  if (response.isSuccessful() && response.body() != null && response.body().getCode() == 200) {
149                      Toast.makeText(OrderDispatchActivity.this, "工单派发成功", Toast.LENGTH_SHORT).show();
150                      finish();
151                  } else {
152                      String msg = response.body() != null ? response.body().getMessage() : "派发失败";
153                      Toast.makeText(OrderDispatchActivity.this, msg, Toast.LENGTH_LONG).show();
154                  }
155              }
156  
157              @Override
158              public void onFailure(Call<ApiResponse<Void>> call, Throwable t) {
159                  btnDispatch.setEnabled(true);
160                  btnDispatch.setText("派发");
161                  Toast.makeText(OrderDispatchActivity.this, "网络错误:" + t.getMessage(), Toast.LENGTH_LONG).show();
162              }
163          });
164      }
165  }
